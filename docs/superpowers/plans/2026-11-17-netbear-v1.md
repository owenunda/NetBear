# NetBear v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** App de bandeja para Windows 11 con un oso pixel art que corre según el tráfico de red, distribuida como `.exe` Native AOT, instalador Inno Setup y zip portable.

**Architecture:** Win32 puro vía `LibraryImport` (sin WinForms). Una ventana oculta recibe timers y mensajes del icono de bandeja. La lógica pura (formateo, cálculo de velocidad, estado y animación, sprites) vive en clases estáticas testeables con xUnit. La parte Win32 (`Native`, `IconRenderer`, `TrayIcon`) se prueba a mano.

**Tech Stack:** C# / .NET 10 (`net10.0-windows`), Native AOT, xUnit, Inno Setup 6, GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-11-17-netbear-design.md`

## Global Constraints

- TargetFramework: `net10.0-windows`. Sin paquetes NuGet en la app (los tests sí usan xUnit).
- `PublishAot=true`. Todo P/Invoke se hace con `LibraryImport` (generado en tiempo de compilación), nunca con `DllImport` con marshalling ni con reflexión.
- Textos de la UI en español. Identificadores de código en inglés.
- Clases `internal`, expuestas a los tests con `InternalsVisibleTo`.
- Velocidades en base 1024: `B/s` sin decimales; `KB/s`, `MB/s` y `GB/s` con un decimal; separador decimal `.`.
- Umbral de carrera: 10 KB/s (10240 B/s). Carrera a velocidad máxima desde 10 MB/s (10485760 B/s). Cuadros de 200 ms (lento) a 40 ms (rápido).
- Hibernando: 500 ms por cuadro. Quieto: 250 ms por cuadro.
- Clave de autostart: `HKCU\Software\Microsoft\Windows\CurrentVersion\Run`, valor `NetBear`.
- URL del repositorio: `https://github.com/owenunda/NetBear`.
- Raíz del repo: `C:\Users\owenu\NetBear`. Todos los comandos se ejecutan desde ahí.

```csharp
using System.Globalization;

namespace NetBear;

internal static class SpeedFormatter
{
    static readonly string[] Units = ["KB/s", "MB/s", "GB/s"];

    public static string Format(double bytesPerSecond)
    {
        if (bytesPerSecond < 1024)
            return $"{(long)Math.Max(0, bytesPerSecond)} B/s";

        double value = bytesPerSecond / 1024;
        int unit = 0;
        while (value >= 1024 && unit < Units.Length - 1)
        {
            value /= 1024;
            unit++;
        }
        return value.ToString("0.0", CultureInfo.InvariantCulture) + " " + Units[unit];
    }
}
```

```csharp
using System.Diagnostics;
using System.Net.NetworkInformation;

namespace NetBear;

internal readonly record struct Speed(double DownBps, double UpBps)
{
    public double Total => DownBps + UpBps;
}

internal static class SpeedCalculator
{
    public static Speed Compute(
        IReadOnlyDictionary<string, (long Rx, long Tx)> previous,
        IReadOnlyDictionary<string, (long Rx, long Tx)> current,
        double elapsedSeconds)
    {
        if (elapsedSeconds <= 0) return default;

        long rx = 0, tx = 0;
        foreach (var (id, now) in current)
        {
            if (!previous.TryGetValue(id, out var before)) continue;
            // Un contador que baja significa que el adaptador se reconectó: cuenta como 0.
            rx += Math.Max(0, now.Rx - before.Rx);
            tx += Math.Max(0, now.Tx - before.Tx);
        }
        return new Speed(rx / elapsedSeconds, tx / elapsedSeconds);
    }
}

internal sealed class NetSpeedMonitor
{
    Dictionary<string, (long Rx, long Tx)> _previous = Sample();
    long _previousTimestamp = Stopwatch.GetTimestamp();

    public Speed Update()
    {
        var current = Sample();
        long now = Stopwatch.GetTimestamp();
        var speed = SpeedCalculator.Compute(_previous, current, Stopwatch.GetElapsedTime(_previousTimestamp, now).TotalSeconds);
        _previous = current;
        _previousTimestamp = now;
        return speed;
    }

    public static bool IsConnected() => NetworkInterface.GetIsNetworkAvailable();

    static Dictionary<string, (long Rx, long Tx)> Sample()
    {
        var counters = new Dictionary<string, (long Rx, long Tx)>();
        NetworkInterface[] adapters;
        try { adapters = NetworkInterface.GetAllNetworkInterfaces(); }
        catch (NetworkInformationException) { return counters; }

        foreach (var adapter in adapters)
        {
            if (adapter.OperationalStatus != OperationalStatus.Up) continue;
            if (adapter.NetworkInterfaceType is NetworkInterfaceType.Loopback or NetworkInterfaceType.Tunnel) continue;
            try
            {
                var stats = adapter.GetIPStatistics();
                counters[adapter.Id] = (stats.BytesReceived, stats.BytesSent);
            }
            catch (NetworkInformationException) { } // se ignora este adaptador en esta muestra
        }
        return counters;
    }
}
```

```csharp
namespace NetBear;

internal enum BearState { Hibernating, Idle, Running }

internal static class BearAnimator
{
    const double RunThresholdBps = 10 * 1024;
    const double MaxSpeedBps = 10 * 1024 * 1024;
    const int SlowestRunMs = 200;
    const int FastestRunMs = 40;

    public static BearState GetState(bool connected, double totalBps) =>
        !connected ? BearState.Hibernating
        : totalBps < RunThresholdBps ? BearState.Idle
        : BearState.Running;

    public static int GetFrameIntervalMs(BearState state, double totalBps) => state switch
    {
        BearState.Hibernating => 500,
        BearState.Idle => 250,
        _ => RunIntervalMs(totalBps),
    };

    // Escala logarítmica: cada x10 de velocidad acelera lo mismo, así 100 KB/s y 50 MB/s se distinguen.
    static int RunIntervalMs(double totalBps)
    {
        double clamped = Math.Clamp(totalBps, RunThresholdBps, MaxSpeedBps);
        double t = Math.Log(clamped / RunThresholdBps) / Math.Log(MaxSpeedBps / RunThresholdBps);
        return (int)Math.Round(SlowestRunMs + (FastestRunMs - SlowestRunMs) * t);
    }
}
```

```csharp
// E  ojo            N  nariz      Z  zzz
internal static class Sprites
{
    public const int Size = 16;

    const string Empty = "................";

    static readonly string[] Body =
    [
        Empty,
        Empty,
        Empty,
        "..........OO.OO.",
        ".........OBBOBBO",
        "..OOOOOOOBBBBBBO",
        ".OBBBBBBBBBBBEBO",
        "OBBBBBBBBBBBBLNO",
        "OBBBBBBBBBBBLLLO",
        "OBBBBBBBBBBBOOO.",
        ".OBBBBBBBBBBO...",
        ".OBBOOOOOOBBO...",
    ];

    static readonly string[] BlinkingBody = [.. Body[..6], ".OBBBBBBBBBBBOBO", .. Body[7..]];

    static string[] WithLegs(string[] body, string upper, string lower) => [.. body, upper, lower, Empty, Empty];

    static readonly string[][] Running =
    [
        WithLegs(Body, "OBO........OBO..", "OO..........OO.."), // estirado
        WithLegs(Body, "OBO......OBO....", "OO.......OO....."), // empuja
        WithLegs(Body, "...OBO..OBO.....", "....OO..OO......"), // cruzado

        WithLegs(Body, "..OBO......OBO..", "..OO........OO.."), // alcanza
    ];

    static readonly string[] Standing = WithLegs(Body, "..OBO.....OBO...", "..OOO.....OOO...");
    static readonly string[] Blinking = WithLegs(BlinkingBody, "..OBO.....OBO...", "..OOO.....OOO...");

    static readonly string[][] Idle =
        [Standing, Standing, Standing, Standing, Standing, Standing, Standing, Standing, Standing, Standing, Standing, Blinking];
```

```csharp
using static NetBear.Native;

namespace NetBear;

internal static unsafe class IconRenderer
{
    // Escala por vecino más cercano: el pixel art queda nítido a cualquier tamaño de icono.
    public static nint Create(string[] frame, IReadOnlyDictionary<char, uint> palette, int size)
    {
        var header = new BITMAPINFOHEADER
        {
            biSize = (uint)sizeof(BITMAPINFOHEADER),
            biWidth = size,
            biHeight = -size, // negativo = filas de arriba hacia abajo
            biPlanes = 1,
            biBitCount = 32,
        };

        void* bits;
        nint screen = GetDC(0);
        nint color = CreateDIBSection(screen, &header, 0, &bits, 0, 0);
        ReleaseDC(0, screen);

        var pixels = new Span<uint>(bits, size * size);
        for (int y = 0; y < size; y++)
        {
            string row = frame[y * Sprites.Size / size];
            for (int x = 0; x < size; x++)
            {
                char c = row[x * Sprites.Size / size];
                pixels[y * size + x] = c == '.' ? 0u : palette[c];
            }
        }

        // La máscara es obligatoria pero, con color de 32 bits con alfa, Windows usa el alfa.
        var maskBits = new byte[(size + 15) / 16 * 2 * size];
        nint mask;
        fixed (byte* m = maskBits) mask = CreateBitmap(size, size, 1, 1, m);

        var info = new ICONINFO { fIcon = 1, hbmMask = mask, hbmColor = color };
        nint icon = CreateIconIndirect(&info);
        DeleteObject(color);
        DeleteObject(mask);
        return icon;
    }
}
```

```csharp
using System.Runtime.InteropServices;
using static NetBear.Native;

namespace NetBear;

internal sealed unsafe class TrayIcon : IDisposable
{
    const string GitHubUrl = "https://github.com/owenunda/NetBear";
    const uint TrayCallback = WM_APP + 1;
    const nuint SampleTimerId = 1, FrameTimerId = 2;
    const nuint CmdAutostart = 1, CmdGitHub = 2, CmdExit = 3;

    // El WndProc es estático (requisito de UnmanagedCallersOnly); solo hay una instancia por proceso.
    static TrayIcon? s_instance;
    static readonly uint s_taskbarCreated = RegisterWindowMessageW("TaskbarCreated");

    readonly nint _hwnd;
    readonly NetSpeedMonitor _monitor = new();
    readonly Dictionary<string[], nint> _icons = new(ReferenceEqualityComparer.Instance);
    readonly int _iconSize = GetSystemMetrics(SM_CXSMICON);
    bool _lightTaskbar = Theme.IsTaskbarLight();
    BearState _state;
    int _frame, _frameMs;
    string _speedText = "";

    public TrayIcon()
    {
        s_instance = this;
        nint instance = GetModuleHandleW(null);
        fixed (char* name = "NetBear")
        {
            var wc = new WNDCLASSEXW
            {
                cbSize = (uint)sizeof(WNDCLASSEXW),
                lpfnWndProc = &WndProc,
                hInstance = instance,
                lpszClassName = name,
            };
            RegisterClassExW(&wc);
            // Ventana normal que nunca se muestra: las de solo mensajes no reciben TaskbarCreated ni WM_SETTINGCHANGE.
            _hwnd = CreateWindowExW(0, name, name, 0, 0, 0, 0, 0, 0, 0, instance, 0);
        }

        Refresh();
        Notify(NIM_ADD);
        SetTimer(_hwnd, SampleTimerId, 1000, 0);
    }

    public void Run()
    {
        MSG msg;
        while (GetMessageW(&msg, 0, 0, 0) > 0)
        {
            TranslateMessage(&msg);
            DispatchMessageW(&msg);
        }
    }

    public void Dispose()
    {
        foreach (nint icon in _icons.Values) DestroyIcon(icon);
        _icons.Clear();
    }

    void Refresh()
    {
        var speed = _monitor.Update();
        bool connected = NetSpeedMonitor.IsConnected();
        _state = BearAnimator.GetState(connected, speed.Total);
        _speedText = connected
            ? $"↓ {SpeedFormatter.Format(speed.DownBps)}  ↑ {SpeedFormatter.Format(speed.UpBps)}"
            : "Sin conexión";

        int ms = BearAnimator.GetFrameIntervalMs(_state, speed.Total);
        if (ms != _frameMs)
        {
            _frameMs = ms;
            SetTimer(_hwnd, FrameTimerId, (uint)ms, 0); // reemplaza el timer anterior con el mismo id
        }
    }

    void Notify(uint message)
    {
        var frames = Sprites.For(_state);
        var frame = frames[_frame % frames.Length];
        if (!_icons.TryGetValue(frame, out nint icon))
            _icons[frame] = icon = IconRenderer.Create(frame, Sprites.Palette(_lightTaskbar), _iconSize);

        var data = new NOTIFYICONDATAW
        {
            cbSize = (uint)sizeof(NOTIFYICONDATAW),
            hWnd = _hwnd,
            uID = 1,
            uFlags = NIF_MESSAGE | NIF_ICON | NIF_TIP,
            uCallbackMessage = TrayCallback,
            hIcon = icon,
        };
        string tip = "NetBear\n" + _speedText;
        tip.AsSpan(0, Math.Min(tip.Length, 127)).CopyTo(new Span<char>(data.szTip, 128));
        Shell_NotifyIconW(message, &data);
    }

    void RemoveIcon()
    {
        var data = new NOTIFYICONDATAW { cbSize = (uint)sizeof(NOTIFYICONDATAW), hWnd = _hwnd, uID = 1 };
        Shell_NotifyIconW(NIM_DELETE, &data);
    }

    void ReloadTheme()
    {
        bool light = Theme.IsTaskbarLight();
        if (light == _lightTaskbar) return;
        _lightTaskbar = light;
        Dispose(); // los iconos cacheados usan la paleta anterior
        Notify(NIM_MODIFY);
    }

    void ShowMenu()
    {
        nint menu = CreatePopupMenu();
        AppendMenuW(menu, MF_STRING | MF_GRAYED, 0, _speedText);
        AppendMenuW(menu, MF_SEPARATOR, 0, null);
        AppendMenuW(menu, MF_STRING | (Autostart.IsEnabled() ? MF_CHECKED : 0), CmdAutostart, "Iniciar con Windows");
        AppendMenuW(menu, MF_STRING, CmdGitHub, "Ver en GitHub");
        AppendMenuW(menu, MF_STRING, CmdExit, "Salir");

        GetCursorPos(out POINT point);
        SetForegroundWindow(_hwnd); // sin esto el menú no se cierra al hacer clic afuera
        nuint command = (nuint)TrackPopupMenu(menu, TPM_RETURNCMD | TPM_RIGHTBUTTON | TPM_NONOTIFY, point.X, point.Y, 0, _hwnd, 0);
        PostMessageW(_hwnd, WM_NULL, 0, 0);
        DestroyMenu(menu);

        switch (command)
        {
            case CmdAutostart: Autostart.Set(!Autostart.IsEnabled()); break;
            case CmdGitHub: ShellExecuteW(0, "open", GitHubUrl, null, null, SW_SHOWNORMAL); break;
            case CmdExit: DestroyWindow(_hwnd); break;
        }
    }

    [UnmanagedCallersOnly]
    static nint WndProc(nint hwnd, uint msg, nint wParam, nint lParam)
    {
        // Durante CreateWindowExW todavía no hay _hwnd asignado: esos mensajes van al manejo por defecto.
        if (s_instance is { } self && self._hwnd == hwnd)
        {
            switch (msg)
            {
                case WM_TIMER when (nuint)wParam == SampleTimerId:
                    self.Refresh();
                    self.Notify(NIM_MODIFY);
                    return 0;
                case WM_TIMER when (nuint)wParam == FrameTimerId:
                    self._frame++;
                    self.Notify(NIM_MODIFY);
                    return 0;
                case TrayCallback when (uint)(lParam & 0xFFFF) == WM_RBUTTONUP:
                    self.ShowMenu();
                    return 0;
                case WM_SETTINGCHANGE when lParam != 0 && new string((char*)lParam) == "ImmersiveColorSet":
                    self.ReloadTheme();
                    return 0;
                case WM_DESTROY:
                    self.RemoveIcon();
                    PostQuitMessage(0);
                    return 0;
            }
            if (msg == s_taskbarCreated) // el Explorador se reinició: hay que volver a agregar el icono
            {
                self.Notify(NIM_ADD);
                return 0;
            }
        }
        return DefWindowProcW(hwnd, msg, wParam, lParam);
    }
}
```

````markdown
Un oso en pixel art que vive en la bandeja de Windows y corre según tu velocidad de internet.
````
